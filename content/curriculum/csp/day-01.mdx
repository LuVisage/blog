## 今日模板

完整版本见[代码模板速查手册](/learn/csp/templates)。

```cpp
// 万能开头
#include <bits/stdc++.h>
using namespace std;
int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    return 0;
}

// 结构体排序
struct Stu { int id, score; };
bool cmp(const Stu &x, const Stu &y) {
    if (x.score != y.score) return x.score > y.score;
    return x.id < y.id;
}
sort(a + 1, a + n + 1, cmp);

// gcd / 快速幂
int gcd(int a, int b) { return b ? gcd(b, a % b) : a; }
long long qpow(long long a, long long b, long long mod) {
    long long res = 1;
    while (b) {
        if (b & 1) res = res * a % mod;
        a = a * a % mod;
        b >>= 1;
    }
    return res;
}

// 二分答案框架
bool check(long long mid) { /* 判断 mid 是否可行 */ }
long long l = 0, r = 1e18, ans = 0;
while (l <= r) {
    long long mid = (l + r) >> 1;
    if (check(mid)) { ans = mid; l = mid + 1; }
    else r = mid - 1;
}

// 前缀和
for (int i = 1; i <= n; i++) s[i] = s[i-1] + a[i];   // 区间和 [l,r] = s[r] - s[l-1]
```
