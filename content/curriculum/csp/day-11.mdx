## 学什么

### 1. 树状数组：单点修改 + 区间查询
```cpp
int c[100005], n;
int lowbit(int x) { return x & -x; }
void add(int i, int x) { for (; i <= n; i += lowbit(i)) c[i] += x; }
int sum(int i) { int s = 0; for (; i > 0; i -= lowbit(i)) s += c[i]; return s; }
// 区间和 [l,r] = sum(r) - sum(l-1)
// 区间修改 + 单点查询：对差分数组建树状数组：add(l,x); add(r+1,-x)
```

### 2. 线段树（入门版：单点修改 + 区间和）
```cpp
long long tr[400005];                    // 开 4 倍空间
void update(int o, int l, int r, int p, long long x) {   // a[p] += x
    if (l == r) { tr[o] += x; return; }
    int mid = (l + r) >> 1;
    if (p <= mid) update(o << 1, l, mid, p, x);
    else update(o << 1 | 1, mid + 1, r, p, x);
    tr[o] = tr[o << 1] + tr[o << 1 | 1];
}
long long query(int o, int l, int r, int ql, int qr) {
    if (ql <= l && r <= qr) return tr[o];
    int mid = (l + r) >> 1;
    long long res = 0;
    if (ql <= mid) res += query(o << 1, l, mid, ql, qr);
    if (qr > mid) res += query(o << 1 | 1, mid + 1, r, ql, qr);
    return res;
}
```

### 3. 区间修改需要懒标记（今天必须会）
```cpp
long long tr[400005], lazy[400005];
void pushdown(int o, int l, int r) {
    if (!lazy[o]) return;
    int mid = (l + r) >> 1;
    tr[o<<1] += lazy[o] * (mid - l + 1);  lazy[o<<1] += lazy[o];
    tr[o<<1|1] += lazy[o] * (r - mid);    lazy[o<<1|1] += lazy[o];
    lazy[o] = 0;
}
void update(int o, int l, int r, int ql, int qr, long long x) {  // 区间加
    if (ql <= l && r <= qr) { tr[o] += x * (r - l + 1); lazy[o] += x; return; }
    pushdown(o, l, r);
    int mid = (l + r) >> 1;
    if (ql <= mid) update(o<<1, l, mid, ql, qr, x);
    if (qr > mid) update(o<<1|1, mid+1, r, ql, qr, x);
    tr[o] = tr[o<<1] + tr[o<<1|1];
}
long long query(int o, int l, int r, int ql, int qr) {
    if (ql <= l && r <= qr) return tr[o];
    pushdown(o, l, r);
    int mid = (l + r) >> 1;
    long long res = 0;
    if (ql <= mid) res += query(o<<1, l, mid, ql, qr);
    if (qr > mid) res += query(o<<1|1, mid+1, r, ql, qr);
    return res;
}
```
