```cpp
long long qpow(long long a, long long b, long long mod) {
    long long res = 1;

        if (b & 1) res = res * a % mod;
        a = a * a % mod;
        b >>= 1;
    }
    return res;

bool isPrime(int n) {
    if (n < 2) return false;

}
```

## 10. 图论：Dijkstra / Floyd / Kruskal

```cpp
// Dijkstra 堆优化（单源最短路，边权非负）
vector<pair<int,long long>> g[100005];   // g[u] = {(v, w)}
long long dist[100005]; bool vis[100005];

void dijkstra(int s, int n) {
    for (int i = 1; i <= n; i++) dist[i] = 1e18, vis[i] = false;
    priority_queue<pair<long long,int>, vector<pair<long long,int>>,
                   greater<pair<long long,int>>> pq;
    dist[s] = 0; pq.push({0, s});
    while (!pq.empty()) {
        int u = pq.top().second; long long d = pq.top().first; pq.pop();
        if (vis[u]) continue;
        vis[u] = true;
        for (auto &e : g[u]) {
            int v = e.first; long long w = e.second;
            if (dist[v] > d + w) {
                dist[v] = d + w;
                pq.push({dist[v], v});
            }
        }
    }
}

// Floyd（全源，n <= 500；初始化 dist[i][j]=边权，无边=1e18，dist[i][i]=0）
for (int k = 1; k <= n; k++)
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= n; j++)
            dist[i][j] = min(dist[i][j], dist[i][k] + dist[k][j]);

// Kruskal（最小生成树：排序 + 并查集）
struct Edge { int u, v; long long w; } e[200005];
bool cmp(Edge a, Edge b) { return a.w < b.w; }
sort(e + 1, e + m + 1, cmp);
long long ans = 0; int cnt = 0;
for (int i = 1; i <= m; i++) {
    if (find(e[i].u) != find(e[i].v)) {
        merge(e[i].u, e[i].v);
        ans += e[i].w; cnt++;
        if (cnt == n - 1) break;
    }
}
// cnt < n-1 → 不连通；最大生成树把 cmp 的 < 改成 >
```

## 11. 动态规划

```cpp
// 01 背包：n 个物品，容量 W，w[i] 重量 v[i] 价值，每个最多一次
int dp[1005] = {0};
for (int i = 1; i <= n; i++)
    for (int j = W; j >= w[i]; j--)      // 倒序！
        dp[j] = max(dp[j], dp[j - w[i]] + v[i]);
// 答案 dp[W]

// 完全背包：每个无限次 —— 循环改正序即可
for (int i = 1; i <= n; i++)
    for (int j = w[i]; j <= W; j++)
        dp[j] = max(dp[j], dp[j - w[i]] + v[i]);

// 最大子段和
// dp[i] = max(a[i], dp[i-1] + a[i]);  ans = max(dp[1..n])
// 滚动版：cur = max(a[i], cur + a[i]); ans = max(ans, cur);

// LIS 最长上升子序列 O(n^2)
for (int i = 1; i <= n; i++) {
    dp[i] = 1;
    for (int j = 1; j < i; j++)
        if (a[j] < a[i]) dp[i] = max(dp[i], dp[j] + 1);
}
```

```cpp
// DFS 全排列
int a[15]; bool used[15]; int n;
void dfs(int k) {
    if (k > n) {
        for (int i = 1; i <= n; i++) cout << a[i] << " ";
        cout << "\n"; return;
    }
    for (int i = 1; i <= n; i++) {
        if (!used[i]) {
            used[i] = true; a[k] = i;
            dfs(k + 1);
            used[i] = false;    // 回溯
        }
    }
}

// BFS 网格最短路
int dx[4] = {0, 0, 1, -1}, dy[4] = {1, -1, 0, 0};
int step[1005][1005];      // -1 未访问
void bfs(int sx, int sy) {
    memset(step, -1, sizeof(step));
    queue<pair<int,int>> q;
    step[sx][sy] = 0; q.push({sx, sy});
    while (!q.empty()) {
        int x = q.front().first, y = q.front().second; q.pop();
        for (int i = 0; i < 4; i++) {
            int nx = x + dx[i], ny = y + dy[i];
            if (nx < 1 || nx > n || ny < 1 || ny > m) continue;   // 越界
            if (a[nx][ny] == 0 || step[nx][ny] != -1) continue;   // 墙/已访问
            step[nx][ny] = step[x][y] + 1;
            q.push({nx, ny});
        }
    }
}
// 多源 BFS：初始把全部源点 step=0 并入队
```

## 16. 字符串哈希（双哈希防冲突）

```cpp
const unsigned long long B = 131;
const unsigned long long M1 = 1000000007, M2 = 998244353;
struct Hash {
    int n; string s;
    vector<unsigned long long> h1, h2, pw1, pw2;
    void init(const string &t) {
        s = " " + t; n = (int)t.size();
        h1.assign(n + 1, 0); h2.assign(n + 1, 0);
        pw1.assign(n + 1, 1); pw2.assign(n + 1, 1);
        for (int i = 1; i <= n; i++) {
            h1[i] = (h1[i-1] * B + s[i]) % M1;
            h2[i] = (h2[i-1] * B + s[i]) % M2;
            pw1[i] = pw1[i-1] * B % M1;
            pw2[i] = pw2[i-1] * B % M2;
        }
    }
    pair<unsigned long long, unsigned long long> get(int l, int r) {
        unsigned long long a = (h1[r] + M1 - h1[l-1] * pw1[r-l+1] % M1) % M1;
        unsigned long long b = (h2[r] + M2 - h2[l-1] * pw2[r-l+1] % M2) % M2;
        return {a, b};
    }
};
// 用途：O(1) 判断两子串相等；配合 map<pair,ll> 计数；与二分结合求最长公共前缀
```

## 19. 线段树（单点修改 + 区间和，入门版）

```cpp
long long tr[400005];   // 开 4 倍
void update(int o, int l, int r, int p, long long x) {  // a[p] += x
    if (l == r) { tr[o] += x; return; }
    int mid = (l + r) >> 1;
    if (p <= mid) update(o << 1, l, mid, p, x);
    else update(o << 1 | 1, mid + 1, r, p, x);
    tr[o] = tr[o << 1] + tr[o << 1 | 1];
}
long long query(int o, int l, int r, int ql, int qr) {
    if (ql <= l && r <= qr) return tr[o];
    int mid = (l + r) >> 1;
    long long res = 0;
    if (ql <= mid) res += query(o << 1, l, mid, ql, qr);
    if (qr > mid) res += query(o << 1 | 1, mid + 1, r, ql, qr);
    return res;
}
// 区间最值把 tr[o] = max(tr[o<<1], tr[o<<1|1])；区间修改需懒标记 lazy（进阶再补）
```

```cpp
// dp[u][0]：不选 u 的最优值；dp[u][1]：选 u 的最优值
void dfs(int u, int fa) {
    dp[u][1] = a[u]; dp[u][0] = 0;
    for (int v : g[u]) if (v != fa) {
        dfs(v, u);
        dp[u][0] += max(dp[v][0], dp[v][1]);
        dp[u][1] += dp[v][0];
    }
}
// 树的直径：两遍 BFS/DFS——从任意点找最远点 a，再从 a 找最远点 b，ab 即直径
```
